cmake_minimum_required(VERSION 3.16)
project(snapfs CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(snapfs src/snapfs.cpp)
target_include_directories(snapfs PUBLIC include)
find_package(Threads REQUIRED)
target_link_libraries(snapfs PUBLIC Threads::Threads)
add_executable(snapfsctl src/main.cpp)
target_link_libraries(snapfsctl PRIVATE snapfs)

option(SNAPFS_FUSE "Build FUSE mount (requires libfuse)" OFF)
if(SNAPFS_FUSE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FUSE REQUIRED fuse3)
  add_executable(snapfs-mount src/fuse_main.cpp)
  target_include_directories(snapfs-mount PRIVATE ${FUSE_INCLUDE_DIRS})
  target_link_libraries(snapfs-mount PRIVATE snapfs ${FUSE_LIBRARIES})
  target_compile_options(snapfs-mount PRIVATE ${FUSE_CFLAGS_OTHER})
endif()

option(SNAPFS_TSAN "Build TSan concurrency test" OFF)
enable_testing()
add_executable(test_unit tests/test_unit.cpp)
target_link_libraries(test_unit PRIVATE snapfs)
add_executable(test_crash tests/test_crash.cpp)
target_link_libraries(test_crash PRIVATE snapfs)
add_executable(test_concurrency tests/test_concurrency.cpp)
target_link_libraries(test_concurrency PRIVATE snapfs)
add_test(NAME unit COMMAND test_unit)
add_test(NAME crash COMMAND test_crash)
add_test(NAME concurrency COMMAND test_concurrency)
if(SNAPFS_TSAN)
  add_executable(test_concurrency_tsan tests/test_concurrency.cpp)
  target_link_libraries(test_concurrency_tsan PRIVATE snapfs)
  target_compile_options(test_concurrency_tsan PRIVATE -fsanitize=thread -g)
  target_link_options(test_concurrency_tsan PRIVATE -fsanitize=thread)
  # snapfs lib itself must also be TSan-instrumented for a meaningful run:
  target_compile_options(snapfs PRIVATE -fsanitize=thread -g)
  target_link_options(snapfs PRIVATE -fsanitize=thread)
  add_test(NAME concurrency_tsan COMMAND test_concurrency_tsan)
endif()
